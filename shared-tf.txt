          // ==== title-to-house flight (s02, s03): identical in both ====
          // The title's staff, its zoom and the bead's flight back into the top house, all from film time, so
          // s02's last frame and s03's first draw the same bead (HANDOFFS § S2 to S3). The type layer zooms out
          // 4 % over s02 (power1.out, Lesson 2's); the bead rests on the staff's middle line, riding its living
          // displacement, until FLY0, then flies to the house's centre in F-cold over 1.0 s (power2.inOut) and
          // shrinks into it over the last 0.3 s as the house lights.
          const TF = { s02: L.scene('s02-title'), title: cueF('title'), YB: 696, S: 28, REST: [1060, 640], HOUSE: [1464, 215] }
          TF.zoom = (T) => 1.04 - 0.04 * P(T, TF.s02.start, TF.s02.end - TF.s02.start, 'power1.out')   // a scale (clock-lint: ok)
          TF.zmap = (x, y, T) => { const z = TF.zoom(T); return [960 + z * (x - 960), 540 + z * (y - 540)] }
          TF.CHIME = (k) => TF.title + 0.05 + 0.09 * k             // the sting's five chimes, A3 E4 A4 C#5 E5
          TF.NOTE_X = [560, 760, 960, 1160, 1360]
          TF.FLY0 = cueF('title', 'start', 2.72)                     // the bead lifts off as the staff starts to fade
          TF.FLY = 1.0
          // the staff lines' living displacement at x, with a ripple from each note as it lands (Lesson 2's)
          TF.disp = (x, T) => {
            const u = x / 1920, pin = Math.sin(Math.PI * clamp(u))
            let y = pin * 2.2 * Math.sin(2 * Math.PI * 1.3 * u + 0.9 * T) * (0.65 + 0.35 * Math.sin(0.6 * T))
            TF.NOTE_X.forEach((nx, k) => {
              const tau = T - TF.CHIME(k)
              if (tau <= 0) return
              const dist = Math.abs(x - nx), front = 1100 * tau
              y += 5.5 * Math.exp(-2.4 * tau) * Math.exp(-Math.pow((dist - front) / 150, 2)) * Math.sin((2 * Math.PI * (dist - front)) / 170)
              y += 6 * Math.exp(-Math.pow(dist / 70, 2)) * Math.exp(-7 * tau) * Math.cos(18 * tau)
            })
            return y
          }
          // the bead on screen at film time T once it has lifted off: { x, y, r, lit } (lit is the house's light)
          TF.flight = (T) => {
            const P0 = TF.zmap(TF.REST[0], TF.REST[1] + TF.disp(TF.REST[0], TF.FLY0), TF.FLY0)
            const P2 = TF.HOUSE, peak = P2[1] - 120
            const P1 = [(P0[0] + P2[0]) / 2, 2 * peak - (P0[1] + P2[1]) / 2]
            const u = P(T, TF.FLY0, TF.FLY, 'power2.inOut'), v = 1 - u
            const lit = P(T, TF.FLY0 + TF.FLY - 0.3, 0.3, 'power2.out')
            return {
              x: v * v * P0[0] + 2 * u * v * P1[0] + u * u * P2[0],
              y: v * v * P0[1] + 2 * u * v * P1[1] + u * u * P2[1],
              r: lerp(22 * TF.zoom(TF.FLY0), 22 * 1.2, u) * (1 - lit),
              lit,
            }
          }
          // ==== end title-to-house flight ====
