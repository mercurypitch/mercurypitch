"""Prove that a chunked Merc pass's chunks join: every frame two neighbouring chunks share must be
the same frame.

  python3 tools/merc-joins.py            every boundary of every chunked pass; writes review/merc/joins.md

Neighbouring chunks overlap by three frames (tools/merc-shots.py: a chunk runs from one frame
before its scene to two after it), and the queue dumps each chunk's first and last three frames as
rendered, before encoding (merc-pass.mjs --dump: lab/merc/dump/<chunk>-f<frame>.rgba). So at each
boundary the first frame of chunk k+1, reached by its no-draw pre-roll, is compared with the same
frame rendered inside chunk k's straight run, byte for byte; so are the next two. A chunk that starts
at frame 0 has no pre-roll at all, so its boundary compares a pre-rolled frame with a run straight
from the pass's first frame (m-orange s03 to s04, m-blue s04 to s05). Also compared: the renderer's
root on those frames (the sidecars), and the same frames decoded from the two WebMs and
composited over the room's obsidian, which differ by VP9's coding only (a chunk's first frame is a
key frame; inside a run it is predicted). For scale, one frame against its own encoding differs by
up to about 70 levels at the body's edges, over a few thousand of its 50,625 pixels. Chunks of
different sizes (S14 at 300, S15 at 400) compare their roots only.
"""
import glob
import json
import os
import subprocess

import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)
DUMP = os.path.join(ROOT, 'lab', 'merc', 'dump')


def decoded(chunk_id, ks, w, h):
    """Frames ks of a chunk's WebM, decoded with their alpha, as {k: RGBA bytes} (upright), in one pass."""
    sel = '+'.join(f'eq(n\\,{k})' for k in sorted(ks))
    out = subprocess.run(['ffmpeg', '-hide_banner', '-loglevel', 'error', '-c:v', 'libvpx-vp9',
                          '-i', os.path.join(ROOT, 'assets', 'merc', chunk_id + '-merc.webm'),
                          '-vf', f'select={sel}', '-fps_mode', 'passthrough', '-f', 'rawvideo', '-pix_fmt', 'rgba', '-'],
                         capture_output=True, check=True).stdout
    n = w * h * 4
    assert len(out) == n * len(ks), (chunk_id, ks, len(out))
    return {k: out[i * n:(i + 1) * n] for i, k in enumerate(sorted(ks))}


def diff(a, b):
    """(pixels that differ, the largest channel difference), RGBA as it is."""
    x = np.frombuffer(a, np.uint8).reshape(-1, 4).astype(np.int16)
    y = np.frombuffer(b, np.uint8).reshape(-1, 4).astype(np.int16)
    d = np.abs(x - y).max(axis=1)
    return int((d > 0).sum()), int(d.max())


def seen(a, b):
    """(pixels whose colour differs by more than 2 levels, the largest difference) once both frames are
    composited over the room's obsidian (#0d1117), as a scene shows them."""
    def over(r):
        v = np.frombuffer(r, np.uint8).reshape(-1, 4).astype(np.float32)
        al = v[:, 3:4] / 255
        return np.rint(v[:, :3] * al + np.array([13, 17, 23], np.float32) * (1 - al))
    d = np.abs(over(a) - over(b)).max(axis=1)
    return int((d > 2).sum()), int(d.max())


def main():
    lines = ['# Merc chunk joins', '',
             'Generated by `tools/merc-joins.py` from the chunks in `assets/merc/` and their dumps in',
             '`lab/merc/dump/`. "Rendered" compares the frames as drawn, before encoding, byte for byte;',
             '"decoded" compares the same frames decoded from the two WebMs, composited over the room\'s',
             'obsidian; "root" compares the renderer\'s root on each frame (the sidecars).', '',
             '| Pass | Boundary | Shared frames | Rendered | Root | Decoded, as seen over #0d1117 (pixels off by more than 2 levels, largest difference) |',
             '| ---- | -------- | ------------- | -------- | ---- | ------------------------------------------------------------------------------------ |']
    bad = checked = waiting = 0
    for path in sorted(glob.glob(os.path.join(ROOT, 'merc', 'shots', '*.json'))):
        shot = json.load(open(path))
        cs = sorted(shot.get('chunks', []), key=lambda c: c['from'])
        for a, b in zip(cs, cs[1:]):
            sides = [os.path.join(ROOT, 'assets', 'merc', c['id'] + '-merc.json') for c in (a, b)]
            if not all(os.path.exists(s) for s in sides):
                lines.append(f"| {shot['id']} | {a['scene']} to {b['scene']} | | not rendered yet | | |")
                waiting += 1
                continue
            sa, sb = (json.load(open(s)) for s in sides)
            shared = list(range(b['from'], a['to'] + 1))
            roots = all(sa['root'][f - a['from']] == sb['root'][f - b['from']] for f in shared)
            rendered, dec = [], []
            if a['size'] == b['size']:
                w, h = sa['drawn']
                da = decoded(a['id'], [f - a['from'] for f in shared], w, h)
                db = decoded(b['id'], [f - b['from'] for f in shared], w, h)
                for f in shared:
                    fa, fb = (os.path.join(DUMP, f"{c['id']}-f{f}.rgba") for c in (a, b))
                    if os.path.exists(fa) and os.path.exists(fb):
                        rendered.append(open(fa, 'rb').read() == open(fb, 'rb').read())
                    dec.append(seen(da[f - a['from']], db[f - b['from']]))
            ok = roots and (a['size'] != b['size'] or (rendered and all(rendered)))
            bad += not ok
            checked += 1
            rtxt = ('identical' if all(rendered) else f"{rendered.count(False)} of {len(rendered)} differ") if rendered else \
                ('sizes differ' if a['size'] != b['size'] else 'no dumps')
            dtxt = ', '.join(f'{px} px, {mx}' for px, mx in dec) if dec else '-'
            start0 = ' (chunk k from frame 0: no pre-roll)' if a['from'] == 0 else ''
            lines.append(f"| {shot['id']} | {a['scene']} to {b['scene']}{start0} | {shared[0]} to {shared[-1]} | {rtxt} | "
                         f"{'same' if roots else 'DIFFERENT'} | {dtxt} |")
    verdict = f'{checked - bad} of {checked} boundaries checked join' + (f'; {bad} do not' if bad else '') + \
        (f'; {waiting} not rendered yet' if waiting else '') + '.'
    lines += ['', verdict]
    os.makedirs(os.path.join(ROOT, 'review', 'merc'), exist_ok=True)
    open(os.path.join(ROOT, 'review', 'merc', 'joins.md'), 'w').write('\n'.join(lines) + '\n')
    print('\n'.join(lines))
    return 1 if bad else 0


if __name__ == '__main__':
    raise SystemExit(main())
